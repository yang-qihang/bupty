\section{动机}

\begin{frame}{动机：信息分散在多个平台}
  \begin{itemize}
    \setlength\itemsep{0.9em}
    \item 北邮学生日常要用的信息，分散在好几个互不相干的系统里：
      \begin{itemize}
        \item 查课表、空教室、成绩、考试 —— 移动教务
        \item 看作业与截止时间 —— 教学云
        \item 查班车、节假日、竞赛 —— 公共信息接口
        \item 预约图书馆研讨间 —— 另一个预约系统
      \end{itemize}
    \item 每个系统都要单独打开、单独登录，来回切换很麻烦，还容易漏掉重要信息（比如作业截止、抢约时间）。
  \end{itemize}
\end{frame}

\begin{frame}{动机：把它们聚合起来，用 TUI 呈现}
  \begin{itemize}
    \setlength\itemsep{0.9em}
    \item \textbf{目标}：把上面这些信息聚合到「一个界面」里，减少来回切换。
    \item \textbf{为什么是 TUI（终端界面）而不是 CLI / GUI}：
      \begin{itemize}
        \item 比 CLI（命令行）更友好：不用记一堆参数和命令，方向键 + Enter 就能操作。
        \item 比 GUI（图形界面）更轻量：在终端里就能跑，无需图形环境。
      \end{itemize}
    \item \textbf{为什么用 Bash}：课程要求；整个程序只用 Bash 加少量系统工具（curl 联网、jq 解析数据）实现，没有额外运行时依赖。
  \end{itemize}
\end{frame}

\begin{frame}{每一层做什么}
  \begin{itemize}
    \setlength\itemsep{0.6em}
    \item \textbf{bin/bupty}：程序的「大门」。运行它，它负责找到代码所在位置，并把各个模块加载进来，然后启动界面。
    \item \textbf{tui.sh}：主循环。它一直在等用户按键，把按键交给「当前页面」处理；还负责页面的前进 / 返回（导航栈），以及后台加载数据后的自动刷新。
    \item \textbf{lib/views/*.sh}：一个个「页面」。每个页面只有两件事：\texttt{render\_xxx} 画界面、\texttt{handle\_xxx\_key} 处理按键。
    \item \textbf{lib/*.sh}：被各页面共用的「工具箱」，例如网络请求、解析、保存密码、记住登录状态、画颜色等。
    \item \textbf{学校平台}：真正存放数据的地方。我们不是自己造数据，而是通过 HTTPS 接口去学校服务器「取数」。
  \end{itemize}
\end{frame}

\begin{frame}{核心库逐个看（lib/*.sh）}
  \begin{columns}[T]
    \begin{column}{0.5\textwidth}
      \begin{itemize}
        \setlength\itemsep{0.35em}
        \item \texttt{core.sh}：常量、日志、错误提示、颜色、文本宽度等最基础的工具（不联网）。
        \item \texttt{http.sh}：封装 curl，统一超时与请求头，返回状态码 + 响应体。
        \item \texttt{session.sh}：记住登录后拿到的「令牌」，只放内存、不落盘。
        \item \texttt{creds.sh}：读写账号密码文件（目录 0700、文件 0600，拒绝软链接）。
        \item \texttt{parse.sh}：纯解析函数（周次展开、节次解析、楼名归一），不联网、方便测试。
        \item \texttt{sjd.sh}：移动教务——登录、课表、空教室、成绩、考试。
      \end{itemize}
    \end{column}
    \begin{column}{0.5\textwidth}
      \begin{itemize}
        \setlength\itemsep{0.35em}
        \item \texttt{ucloud.sh}：教学云——统一认证、课程列表、作业。
        \item \texttt{public.sh}：公共接口——班车、节假日、竞赛活动。
        \item \texttt{booking.sh}：研讨间预约——登录、房间与空闲时段、预约、取消、抢约。
        \item \texttt{settings.sh}：读写设置文件（先写临时文件再覆盖，避免写坏）。
        \item \texttt{background.sh}：后台守护——紧急 DDL 弹窗 + 自动抢约。
        \item \texttt{submit.sh}：作业在线提交。
        \item \texttt{ui.sh}：界面原语——清屏、边框、颜色、列表、输入框、按键读取。
      \end{itemize}
    \end{column}
  \end{columns}
\end{frame}

\begin{frame}[fragile]{数据如何流动：JSON 一条道走到底}
  \begin{itemize}
    \setlength\itemsep{0.5em}
    \item \textbf{JSON} 是一种通用的文本格式，用键值对表示数据。各模块之间一律用 JSON 字符串传数据，用命令行工具 \texttt{jq} 读写。
    \item 好处：模块之间只约定「字段名」，不约定「分隔符」，不会因为数据里碰巧出现分隔符而出错。
    \item 页面显示前先做一次「预计算」，把 JSON 整理成一行行文本；真正逐行画界面时只用 Bash 本身，不再反复调用 \texttt{jq}。
  \end{itemize}
  {\footnotesize
  \begin{verbatim}
  # 网络模块返回的 JSON
  {"devId":100888915, "devName":"研讨间314", "free":[{"start":"08:00","end":"11:30"}]}

  # 预计算成一行文本，渲染时直接读
  100888915  研讨间314  08:00-11:30
  \end{verbatim}
  }
\end{frame}

\begin{frame}[fragile]{异步加载：界面始终不「卡住」}
  \begin{itemize}
    \setlength\itemsep{0.5em}
    \item 传统做法：进入页面 → 联网取数据 → 数据到了才显示，网络慢时界面会空白或卡住。
    \item 我们的做法：进入页面后 \textbf{立刻}画出界面框架和「正在加载…」提示，数据在后台悄悄下载，下载完成后自动刷新。
  \end{itemize}
  {\footnotesize
  \begin{verbatim}
  # 进入页面：在后台子进程里拉取，不阻塞界面
  _async_start schedule sjd_fetch_schedule "$acc" "$pwd"

  # 主循环每隔一小段时间检查一次，完成了就刷新
  if _async_poll; then tui_render; fi
  \end{verbatim}
  }
  \begin{itemize}
    \item 渲染用「帧缓冲」：整屏内容先画到临时文件里，再一次性输出到终端，避免一行一行刷出来导致的闪烁。
  \end{itemize}
\end{frame}

\begin{frame}[fragile]{渲染性能：去掉多余的「子进程」}
  \begin{itemize}
    \setlength\itemsep{0.5em}
    \item Bash 里很多写法（比如 \texttt{\$(命令)}、管道）都会额外启动一个「子进程」，每启动一次都要花一点时间。
    \item 如果一个页面有几十上百行，每行都启动几次子进程，界面就会明显变慢。
  \end{itemize}
  {\footnotesize
  \begin{verbatim}
  # 慢：每条都额外启动一个子进程
  fitted="$(core_fit "$cell" "$cw")"

  # 快：用 Bash 内置运算，结果直接写进变量
  _core_fit_compute "$cell" "$cw"
  fitted="$_CORE_FIT_OUT"
  \end{verbatim}
  }
  \begin{itemize}
    \setlength\itemsep{0.5em}
    \item 其它优化：截断超长文字只扫一遍；凭据、活动过滤、班车显示行等只算一次并缓存。
    \item 效果：课表、班车等页面从约 200ms 一帧降到约 10--30ms 一帧。
  \end{itemize}
\end{frame}

\section{设计选择}

\begin{frame}{设计选择：把 Bash 也写得「模块化」}
  \begin{itemize}
    \setlength\itemsep{0.7em}
    \item 不把所有代码塞进一个超大脚本，而是按职责拆成几十个小文件，每个文件只负责一件事。
    \item Bash 里用 \texttt{source} 把其他文件「引入」进来，效果相当于把这些文件拼接在一起，但代码更清晰、更好维护。
    \item 命名约定：纯解析函数（\texttt{lib/parse.sh}）\textbf{不联网}，便于离线单元测试；网络函数成功返回 0、失败返回 1，结果写到标准输出、错误写到一个全局变量。
  \end{itemize}
\end{frame}

\begin{frame}{设计选择：安全}
  \begin{itemize}
    \setlength\itemsep{0.7em}
    \item 保存密码的目录权限设为 0700（只有本人能进）、文件 0600（只有本人能读写），写入时拒绝软链接，密码绝不打印、不写日志。
    \item 登录后服务器发的「令牌」（相当于一张通行证）只存在内存里，不写进磁盘；过期后自动重新登录一次。
    \item 预约系统使用学校的「统一身份认证」（CAS）：在统一登录页登录一次，就能访问多个系统；认证过程全程在同一个会话里顺序完成，一次性参数用后即弃。
  \end{itemize}
\end{frame}

\begin{frame}{设计选择：可配置与后台}
  \begin{itemize}
    \setlength\itemsep{0.7em}
    \item 设置持久化到 \texttt{settings.json}：先把新内容写进一个临时文件，确认无误后再一次性覆盖原文件，中途出错也不会写坏。
    \item \texttt{-{}-background} 后台模式：先检查是否已有另一份在运行、避免重复启动，然后循环拉取作业，对「紧急 DDL」弹窗提醒。
    \item 自动抢约只抢「明天及以后」的时段；遇到永久性错误（如已被禁用）就放弃并通知，临时错误则限次重试。
  \end{itemize}
\end{frame}

\section{测试}

\begin{frame}{测试方法}
  \begin{columns}[T]
    \begin{column}{0.55\textwidth}
      \begin{itemize}
        \item \textbf{单元测试}（\texttt{tests/unit\_*.sh}）：离线、基于预置样例数据，断言解析 / 组装 / 过滤等纯函数的结果。
        \item \textbf{渲染快照}：渲染各页面并断言关键文字（去掉颜色码），验证界面文案正确。
      \end{itemize}
    \end{column}
    \begin{column}{0.45\textwidth}
      \begin{itemize}
        \item \textbf{多窗口尺寸自检}：在 80/100/120/60/140 列宽下断言不溢出、标题在、无横向换行。
        \item \textbf{真实网络集成测试}（只读 + 授权）：登录、拉取、预约 $\to$ 取消，验证协议与实现一致。
        \item \textbf{对抗式评审}：实现 / 测试 / 评审由不同 agent 分别完成，不共享上下文。
      \end{itemize}
    \end{column}
  \end{columns}
\end{frame}

\section{总结}

\begin{frame}{总结}
  \begin{itemize}
    \setlength\itemsep{0.8em}
    \item 用纯 Bash 实现了一个功能完整、交互友好的北邮信息查询 \textbf{TUI}，把分散的信息聚合到一个界面。
    \item 核心经验：模块化拆分、JSON 作为数据契约、异步加载 + 帧缓冲、渲染热循环去子进程、文字宽度只扫一遍。
    \item 通过「单元测试 + 渲染快照 + 多尺寸自检 + 真实网络集成 + 对抗式评审」保证了正确性与界面质量。
    \item 各系统的网络接口均经真实请求逐条核对并文档化。
  \end{itemize}
